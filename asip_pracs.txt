# Upsampling, Downsampling and FFT
# Required Libraries: numpy, matplotlib, scipy

import numpy as np
import matplotlib.pyplot as plt
from scipy.fft import fft

t = np.linspace(0, 1, 100)
signal = np.sin(2 * np.pi * 5 * t)

upsampled = np.repeat(signal, 2)
downsampled = signal[::2]
freq = np.abs(fft(signal))

plt.figure(figsize=(10,6))
plt.subplot(3,1,1); plt.plot(signal); plt.title("Original Signal")
plt.subplot(3,1,2); plt.plot(upsampled); plt.title("Upsampled Signal")
plt.subplot(3,1,3); plt.plot(freq); plt.title("FFT")
plt.show()

# Triangular Signal and Correlation
# Required Libraries: numpy, matplotlib

import numpy as np
import matplotlib.pyplot as plt

t = np.linspace(-1, 1, 100)
tri = 1 - np.abs(t)
seg = tri[20:60]
corr = np.correlate(tri, seg, mode='full')

plt.figure(figsize=(10,5))
plt.subplot(3,1,1); plt.plot(tri); plt.title("Triangular Signal")
plt.subplot(3,1,2); plt.plot(seg); plt.title("Segment")
plt.subplot(3,1,3); plt.plot(corr); plt.title("Correlation")
plt.show()

# Convolution and Template Matching
# Required Libraries: cv2, numpy, matplotlib

import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread('image.jpg', 0)
kernel = np.ones((3,3))/9
conv = cv2.filter2D(img, -1, kernel)
template = img[50:100, 50:100]
res = cv2.matchTemplate(img, template, cv2.TM_CCOEFF)

plt.subplot(1,2,1); plt.imshow(conv, cmap='gray'); plt.title("Convolution")
plt.subplot(1,2,2); plt.imshow(res, cmap='gray'); plt.title("Template Matching")
plt.show()

# Intensity Transformations
# Required Libraries: cv2, numpy, matplotlib

import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread('image.jpg', 0)
log_img = np.log1p(img)
gamma_img = np.power(img/255, 0.5)
hist_eq = cv2.equalizeHist(img)

plt.subplot(2,2,1); plt.imshow(log_img, cmap='gray'); plt.title("Log")
plt.subplot(2,2,2); plt.imshow(gamma_img, cmap='gray'); plt.title("Gamma")
plt.subplot(2,2,3); plt.imshow(hist_eq, cmap='gray'); plt.title("Histogram Equalization")
plt.show()

# Morphological Operations
# Required Libraries: cv2, matplotlib

import cv2
import matplotlib.pyplot as plt

img = cv2.imread('image.jpg', 0)
_, binary = cv2.threshold(img, 127, 255, cv2.THRESH_BINARY)
kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (5,5))

erosion = cv2.erode(binary, kernel)
dilation = cv2.dilate(binary, kernel)

plt.subplot(1,2,1); plt.imshow(erosion, cmap='gray'); plt.title("Erosion")
plt.subplot(1,2,2); plt.imshow(dilation, cmap='gray'); plt.title("Dilation")
plt.show()
